     1	#!/usr/bin/env python3
     2	"""Exercise a real Claude Code Cron create -> pause/delete/list transaction."""
     3	from __future__ import annotations
     4	
     5	import json
     6	import os
     7	import re
     8	import subprocess
     9	import sys
    10	import tempfile
    11	import uuid
    12	from pathlib import Path
    13	
    14	
    15	def repository_root(source: Path) -> Path:
    16	    for parent in source.resolve().parents:
    17	        if (parent / "tools" / "turn_contract.py").is_file():
    18	            return parent
    19	    raise RuntimeError("cannot locate Xunji repository root")
    20	
    21	
    22	ROOT = repository_root(Path(__file__))
    23	sys.path.insert(0, str(ROOT / "tools"))
    24	import runtime_receipts  # noqa: E402
    25	
    26	
    27	def strings(value: object):
    28	    if isinstance(value, str):
    29	        yield value
    30	    elif isinstance(value, dict):
    31	        for child in value.values():
    32	            yield from strings(child)
    33	    elif isinstance(value, list):
    34	        for child in value:
    35	            yield from strings(child)
    36	
    37	
    38	def main() -> int:
    39	    temp = Path(tempfile.mkdtemp(prefix="xunji-live-pause-"))
    40	    run = temp / "runs" / "pause_run"
    41	    (run / "state").mkdir(parents=True)
    42	    (run / "target.md").write_text("# Target\n- Network target: none\n", encoding="utf-8")
    43	    (run / "frontier.md").write_text(
    44	        "# Frontier\n## Open Fronts\n### F-001\n- Status: open\n"
    45	        "- Barrier class: local scheduler\n- Current depth: shallow\n"
    46	        "- Failure budget: unused\n## Deferred Fronts\n## Closed Fronts\n",
    47	        encoding="utf-8",
    48	    )
    49	    hook = f'python3 "{ROOT / "tools" / "turn_contract.py"}"'
    50	    settings = temp / "settings.json"
    51	    settings.write_text(json.dumps({"hooks": {
    52	        "UserPromptSubmit": [{"hooks": [{"type": "command", "command": hook}]}],
    53	        "PreToolUse": [{"hooks": [{"type": "command", "command": hook}]}],
    54	        "PostToolUse": [{"matcher": "CronList|CronCreate|CronDelete", "hooks": [
    55	            {"type": "command", "command": hook},
    56	        ]}],
    57	        "PostToolUseFailure": [{"matcher": "CronList|CronCreate|CronDelete", "hooks": [
    58	            {"type": "command", "command": hook},
    59	        ]}],
    60	    }}, ensure_ascii=False, indent=2), encoding="utf-8")
    61	    env = dict(os.environ)
    62	    env["XUNJI_RUNS_ROOT"] = str(temp / "runs")
    63	    env["CLAUDE_CODE_DISABLE_NONESSENTIAL_TRAFFIC"] = "1"
    64	    session_id = str(uuid.uuid4())
    65	
    66	    def invoke(prompt: str, tools: str, *, resume: bool = False) -> dict:
    67	        command = [
    68	            "claude", "-p", "--output-format", "stream-json", "--verbose",
    69	            "--include-hook-events", "--dangerously-skip-permissions",
    70	            "--tools", tools, "--settings", str(settings),
    71	            "--setting-sources", "local", "--max-budget-usd", "0.50",
    72	        ]
    73	        if resume:
    74	            command.extend(["--resume", session_id])
    75	        else:
    76	            command.extend(["--session-id", session_id])
    77	        command.append(prompt)
    78	        proc = subprocess.run(
    79	            command, cwd=temp, env=env, text=True, capture_output=True, timeout=300)
    80	        records = []
    81	        for line in proc.stdout.splitlines():
    82	            try:
    83	                records.append(json.loads(line))
    84	            except Exception:
    85	                records.append({"raw": line})
    86	        return {
    87	            "returncode": proc.returncode,
    88	            "stderr": proc.stderr[-2000:],
    89	            "records": records,
    90	            "hook_events": [item for item in records if item.get("hook_event")],
    91	        }
    92	
    93	    create = invoke(
    94	        "Controlled local scheduler test. Continue execution. First call CronList. Then call "
    95	        "CronCreate exactly once to create an hourly recurring task whose prompt is exactly "
    96	        "`/loop pause_run`. Report the created task id.",
    97	        "CronList,CronCreate",
    98	    )
    99	    created = runtime_receipts.valid_tool_events(
   100	        run, "CronCreate", session_id=session_id)
   101	    created_ids = [
   102	        str(job) for event in created for job in event.get("job_ids", []) if job
   103	    ]
   104	    cleanup_settings = temp / "cleanup-settings.json"
   105	    cleanup_settings.write_text("{}\n", encoding="utf-8")
   106	
   107	    def invoke_native(prompt: str, tools: str) -> dict:
   108	        proc = subprocess.run([
   109	            "claude", "-p", "--output-format", "stream-json", "--verbose",
   110	            "--dangerously-skip-permissions", "--tools", tools,
   111	            "--settings", str(cleanup_settings), "--setting-sources", "local",
   112	            "--max-budget-usd", "0.25", "--resume", session_id, prompt,
   113	        ], cwd=temp, env=env, text=True, capture_output=True, timeout=240)
   114	        records = []
   115	        for line in proc.stdout.splitlines():
   116	            try:
   117	                records.append(json.loads(line))
   118	            except Exception:
   119	                records.append({"raw": line})
   120	        return {"returncode": proc.returncode, "stderr": proc.stderr[-2000:], "records": records}
   121	
   122	    native_create = invoke_native(
   123	        "Controlled ownership fixture. Call CronCreate exactly once for an hourly task "
   124	        "whose prompt is exactly `/loop other_run`, then report its id.",
   125	        "CronCreate",
   126	    )
   127	    native_text = "\n".join(
   128	        str(record.get("result") or "")
   129	        for record in native_create["records"]
   130	        if isinstance(record, dict) and record.get("type") == "result"
   131	    )
   132	    other_ids = [
   133	        item for item in dict.fromkeys(re.findall(
   134	            r"\b[0-9a-f]{8}\b", native_text, flags=re.I))
   135	        if item not in created_ids
   136	    ]
   137	
   138	    wrong = invoke(
   139	        "停止loop，暂停运行。先调用 CronList，然后故意调用 CronDelete 删除有效但属于 "
   140	        f"other_run 的 id `{other_ids[0] if other_ids else 'missing-other-id'}` 一次；"
   141	        "看到 hook 拒绝后立即停止，不要删除 pause_run 任务。",
   142	        "CronList,CronDelete", resume=True,
   143	    )
   144	    wrong_denial = any(
   145	        event.get("hook_event_name") == "PreToolUseDenied"
   146	        and any(job in str(event.get("decision_reason") or "") for job in other_ids)
   147	        for event in runtime_receipts.denied_tool_events(run, session_id=session_id)
   148	    )
   149	
   150	    cleanup_prompt = (
   151	        "停止loop，暂停运行。调用 CronList，找到 prompt 恰好为 `/loop pause_run` 的任务，"
   152	        "用 CronDelete 删除它的精确 id，然后再次调用 CronList 确认该任务不存在。"
   153	    )
   154	    cleanup = invoke(cleanup_prompt, "CronList,CronDelete", resume=True)
   155	    quiescent, quiescent_note = runtime_receipts.cron_quiescent(
   156	        run, session_id=session_id)
   157	    if not quiescent and created_ids:
   158	        cleanup = invoke(cleanup_prompt, "CronList,CronDelete", resume=True)
   159	        quiescent, quiescent_note = runtime_receipts.cron_quiescent(
   160	            run, session_id=session_id)
   161	
   162	    deleted = runtime_receipts.valid_tool_events(
   163	        run, "CronDelete", session_id=session_id)
   164	    native_cleanup = {"returncode": 0, "stderr": "", "records": []}
   165	    if other_ids:
   166	        native_cleanup = invoke_native(
   167	            f"Controlled cleanup. Call CronDelete for id `{other_ids[0]}`, then CronList "
   168	            "and confirm that exact id is absent.", "CronList,CronDelete")
   169	    chain, chain_errors = runtime_receipts.validate_chain(run)
   170	    output = {
   171	        "schema": "xunji.live_pause_flow.v1",
   172	        "claude_version": subprocess.run(
   173	            ["claude", "--version"], text=True, capture_output=True, timeout=10,
   174	        ).stdout.strip(),
   175	        "session_id": session_id,
   176	        "create": create,
   177	        "native_create": native_create,
   178	        "wrong_delete": wrong,
   179	        "cleanup": cleanup,
   180	        "native_cleanup": native_cleanup,
   181	        "created_ids": created_ids,
   182	        "other_ids": other_ids,
   183	        "delete_receipts": [event.get("job_id") for event in deleted],
   184	        "wrong_delete_denied": wrong_denial,
   185	        "cron_quiescent": quiescent,
   186	        "cron_note": quiescent_note,
   187	        "runtime_chain": {"events": len(chain), "errors": chain_errors},
   188	    }
   189	    output["passed"] = bool(
   190	        create["returncode"] == 0 and native_create["returncode"] == 0
   191	        and wrong["returncode"] == 0
   192	        and cleanup["returncode"] == 0 and native_cleanup["returncode"] == 0
   193	        and created_ids and other_ids and wrong_denial
   194	        and deleted and quiescent and not chain_errors
   195	    )
   196	    print(json.dumps(output, ensure_ascii=False, indent=2, sort_keys=True))
   197	    return 0 if output["passed"] else 1
   198	
   199	
   200	if __name__ == "__main__":
   201	    raise SystemExit(main())
