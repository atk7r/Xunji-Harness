   384	            if not runtime_receipts._evidence_hash(prompt) or missing:
   385	                return (

   388	                    + ", ".join(missing or ["EVIDENCE_INDEX"]))
   389	            return ""
   390	        assignment, front = runtime_receipts._assignment_fields(prompt)
   391	        if not assignment or not front:
   392	            return "Agent Board 强制：Agent prompt 必须包含 XUNJI_ASSIGNMENT=A-... 和 XUNJI_FRONT=F-...。"
   393	        if not _assignment_exists(run_dir, assignment, front):

   414	                    "且更新时间必须晚于 Agent 返回。"

   417	    if tool == "Bash":

   423	            )
   424	    if tool in {"WebFetch"}:
   425	        return "Agent Board 强制：真实 Agent 回执覆盖两个不同 front 前，Root 不得继续目标 WebFetch。"
   426	    return ""

   429	def _context_message(contract: dict, run_dir: Path) -> str:
   430	    mode = contract.get("mode")
   431	    if mode == EXPLAIN:
   432	        return "[Xunji turn mode: EXPLAIN_ONLY] 只回答操作者问题；可读文件，不修改、不探测、不派 Agent；本回合无需 Coda。"
   433	    if mode == PAUSE:
   434	        return "[Xunji turn mode: PAUSED_BY_OPERATOR] 保留所有 open fronts；先 CronList/CronDelete，禁止继续渗透；本回合无需 Coda，也不得写 completion marker。"
   435	    state = run_model.summary(run_dir)
   436	    fanout = (
   437	        " fanout_required=true (open/probing/working/type-A 均为 active); "
   438	        "先真实调用至少两个不同 front 的 Agent。"
   439	        if state.get("fanout_required") else ""
   440	    )
   441	    return "[Xunji turn mode: EXECUTE] 按 run 状态推进，最后写唯一具体 Coda。" + fanout

   444	def handle_event(event: dict, run_dir: Path | None = None) -> dict | None:
   445	    run_dir = run_dir or find_active_run(RUNS, within_sec=6 * 60 * 60)
   446	    if run_dir is None:
   447	        if str(event.get("hook_event_name") or "") == "PreToolUse" \
   448	                and str(event.get("tool_name") or "") == "CronCreate":
   449	            return _deny("Xunji CronCreate 前必须先 setup/set-active 一个 run，再用当前回合 CronList 证明单实例。")
   450	        return None
   451	    hook = str(event.get("hook_event_name") or "")
   452	    if hook == "UserPromptSubmit":
   453	        prompt = str(event.get("prompt") or "")
   454	        if INTERNAL_PROMPT_RE.search(prompt):
   455	            contract = load_contract(
   456	                run_dir, session_id=str(event.get("session_id") or ""))
   457	            if not contract:
   458	                return None
   459	            return {
   460	                "hookSpecificOutput": {
   461	                    "hookEventName": "UserPromptSubmit",
   462	                    "additionalContext": _context_message(contract, run_dir),
   463	                }
   464	            }
   465	        contract = write_contract(run_dir, event)
   466	        return {
   467	            "hookSpecificOutput": {
   468	                "hookEventName": "UserPromptSubmit",
   469	                "additionalContext": _context_message(contract, run_dir),
   470	            }
   471	        }
   472	    if hook == "PreToolUse":
   473	        contract = load_contract(run_dir, session_id=str(event.get("session_id") or ""))
   474	        reason = evaluate_pretool(run_dir, event, contract)
   475	        if reason:
   476	            receipt_event = dict(event)
   477	            receipt_event.update({
   478	                "hook_event_name": "PreToolUseDenied",
   479	                "xunji_decision": "deny",
   480	                "xunji_reason": reason,
   481	                "xunji_target_action": _denial_is_target_action(event, reason),
   482	            })
   483	            runtime_receipts.append_hook_event(run_dir, receipt_event)
   484	            return _deny(reason)
   485	        return None
   486	    if hook in {"PostToolUse", "PostToolUseFailure", "SubagentStart", "SubagentStop"}:
   487	        tool_name = str(event.get("tool_name") or "")
   488	        command = str((event.get("tool_input") or {}).get("command") or "") \
   489	            if isinstance(event.get("tool_input"), dict) else ""
   490	        target_action = _is_target_action(event)
   491	        if hook in {"SubagentStart", "SubagentStop"} or tool_name in {
   492	            "Agent", "CronCreate", "CronDelete", "CronList",
   493	        } or (tool_name == "Bash" and "peer_review.py" in command) or target_action:
   494	            receipt_event = dict(event)
   495	            receipt_event["xunji_target_action"] = target_action
   496	            runtime_receipts.append_hook_event(run_dir, receipt_event)
   497	        return None
   498	    return None
