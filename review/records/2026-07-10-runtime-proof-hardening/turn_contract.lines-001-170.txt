    85	def _atomic_json(path: Path, value: dict) -> None:
    86	    path.parent.mkdir(parents=True, exist_ok=True)
    87	    fd, raw = tempfile.mkstemp(prefix=path.name + ".", suffix=".tmp", dir=path.parent)
    88	    try:
    89	        with os.fdopen(fd, "w", encoding="utf-8") as handle:
    90	            json.dump(value, handle, ensure_ascii=False, indent=2, sort_keys=True)
    91	            handle.write("\n")
    92	        os.chmod(raw, 0o600)
    93	        os.replace(raw, path)
    94	    finally:
    95	        try:
    96	            os.unlink(raw)
    97	        except FileNotFoundError:
    98	            pass

   109	def classify_prompt(prompt: str, *, active_run: bool = True) -> str:
   110	    prompt = prompt or ""
   111	    if not active_run:
   112	        return NORMAL
   113	    if PAUSE_RE.search(prompt):
   114	        return PAUSE
   115	    if EXPLAIN_RE.search(prompt):
   116	        return EXPLAIN
   117	    if EXECUTE_RE.search(prompt):
   118	        return EXECUTE
   119	    if re.search(r"[?？]\s*$", prompt) or re.match(r"^\s*(?:怎么|如何|是否|什么|哪|谁)", prompt):
   120	        return EXPLAIN
   121	    return EXPLAIN

   123	
   124	def write_contract(run_dir: Path, event: dict) -> dict:
   125	    prompt = str(event.get("prompt") or "")
   126	    mode = classify_prompt(prompt, active_run=True)
   127	    now = time.time()
   128	    contract = {
   129	        "schema": SCHEMA,
   130	        "mode": mode,
   131	        "session_id": str(event.get("session_id") or ""),
   132	        "transcript_path": str(event.get("transcript_path") or ""),
   133	        "prompt_sha256": hashlib.sha256(prompt.encode("utf-8", "replace")).hexdigest(),
   134	        "prompt_excerpt": prompt[:500],
   135	        "memory_approved": bool(MEMORY_APPROVAL_RE.search(prompt)),
   136	        "fanout_override": bool(re.search(r"(?:明确)?允许串行|不要使用\s*(?:Agent|子代理)|serial override", prompt, re.I)),
   137	        "updated_at": now,
   138	    }
   139	    _atomic_json(contract_path(run_dir), contract)
   140	    if mode == PAUSE:
   141	        _atomic_json(run_status_path(run_dir), {
   142	            "schema": SCHEMA,
   143	            "status": "paused_by_operator",
   144	            "session_id": contract["session_id"],
   145	            "updated_at": now,
   146	            "reason": "operator prompt requested pause; open fronts remain open",
   147	        })
   148	    elif mode == EXECUTE:
   149	        _atomic_json(run_status_path(run_dir), {
   150	            "schema": SCHEMA,
   151	            "status": "active",
   152	            "session_id": contract["session_id"],
   153	            "updated_at": now,
   154	            "reason": "operator prompt requested execution/resume",
   155	        })
   156	    return contract

   158	
   159	def load_contract(run_dir: Path, *, session_id: str = "") -> dict:
   160	    try:
   161	        data = json.loads(contract_path(run_dir).read_text(encoding="utf-8", errors="replace"))
   162	    except Exception:
   163	        return {}
   164	    if data.get("schema") != SCHEMA:
   165	        return {}
   166	    if session_id and str(data.get("session_id") or "") != session_id:
   167	        return {}
   168	    try:
   169	        if time.time() - float(data.get("updated_at") or 0) > STALE_SECONDS:
   170	            return {}
