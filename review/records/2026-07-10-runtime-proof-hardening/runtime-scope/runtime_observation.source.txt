     1	#!/usr/bin/env python3
     2	"""Exercise installed hook entrypoints against a temporary active run."""
     3	from __future__ import annotations
     4	
     5	import hashlib
     6	import json
     7	import os
     8	import subprocess
     9	import sys
    10	import tempfile
    11	from datetime import datetime, timezone
    12	from pathlib import Path
    13	
    14	
    15	def repository_root(source: Path) -> Path:
    16	    for parent in source.resolve().parents:
    17	        if (parent / "tools" / "turn_contract.py").is_file():
    18	            return parent
    19	    raise RuntimeError("cannot locate repository root containing tools/turn_contract.py")
    20	
    21	
    22	ROOT = repository_root(Path(__file__))
    23	TOOLS = ROOT / "tools"
    24	sys.path.insert(0, str(TOOLS))
    25	
    26	import runtime_receipts  # noqa: E402
    27	import workers  # noqa: E402
    28	
    29	
    30	def main() -> int:
    31	    temp = Path(tempfile.mkdtemp(prefix="xunji-runtime-observation-"))
    32	    runs = temp / "runs"
    33	    run = runs / "observed_run"
    34	    (run / "state").mkdir(parents=True)
    35	    (run / "agents").mkdir()
    36	    (run / "target.md").write_text("# Target\n", encoding="utf-8")
    37	    (run / "frontier.md").write_text(
    38	        "# Frontier\n## Open Fronts\n"
    39	        + "".join(
    40	            f"### F-{index:03d}\n- Status: open\n- Barrier class: {barrier}\n"
    41	            "- Current depth: shallow\n- Failure budget: unused\n"
    42	            for index, barrier in enumerate(("app", "auth", "network", "routing"), 1)
    43	        )
    44	        + "## Deferred Fronts\n## Closed Fronts\n",
    45	        encoding="utf-8",
    46	    )
    47	    assignments = [
    48	        {"agent": "A-web-001", "front": "F-001", "status": "assigned"},
    49	        {"agent": "A-auth-001", "front": "F-002", "status": "assigned"},
    50	    ]
    51	    (run / "state" / "assignments.json").write_text(
    52	        json.dumps({"schema": 1, "assignments": assignments}), encoding="utf-8")
    53	    for item in assignments:
    54	        (run / "agents" / f"{item['agent']}.md").write_text(
    55	            f"# Agent\n- Role: verify\n- Assigned front: {item['front']}\n- Status: assigned\n",
    56	            encoding="utf-8",
    57	        )
    58	
    59	    transcript = temp / "transcript.jsonl"
    60	    tool_ids = [
    61	        "agent-tool-1", "agent-tool-2", "cron-list-1", "cron-create-1",
    62	        "cron-list-2", "cron-delete-1", "cron-list-3", "review-tool-1",
    63	        "completion-weak-1", "completion-structured-1",
    64	    ]
    65	    transcript.write_text("\n".join(tool_ids) + "\n", encoding="utf-8")
    66	    env = dict(os.environ)
    67	    env["XUNJI_RUNS_ROOT"] = str(runs)
    68	    settings = json.loads((ROOT / ".claude" / "settings.json").read_text(encoding="utf-8"))
    69	    session_start_commands = [
    70	        str(item.get("command") or "")
    71	        for group in settings.get("hooks", {}).get("SessionStart", [])
    72	        if isinstance(group, dict)
    73	        for item in group.get("hooks", [])
    74	        if isinstance(item, dict)
    75	    ]
    76	
    77	    def hook(event: dict) -> dict:
    78	        event.setdefault("transcript_path", str(transcript))
    79	        proc = subprocess.run(
    80	            [sys.executable, str(TOOLS / "turn_contract.py")],
    81	            input=json.dumps(event), text=True, capture_output=True, env=env, timeout=15,
    82	        )
    83	        if proc.returncode != 0:
    84	            raise RuntimeError(proc.stderr or f"hook returncode={proc.returncode}")
    85	        return json.loads(proc.stdout) if proc.stdout.strip() else {}
    86	
    87	    def prompt(text: str, session: str) -> dict:
    88	        return hook({"hook_event_name": "UserPromptSubmit", "session_id": session, "prompt": text})
    89	
    90	    def pre(tool: str, tool_input: dict, session: str = "s-exec") -> dict:
    91	        return hook({
    92	            "hook_event_name": "PreToolUse", "session_id": session,
    93	            "tool_name": tool, "tool_input": tool_input,
    94	        })
    95	
    96	    def post(tool: str, tool_id: str, tool_input: dict, response: object) -> None:
    97	        hook({
    98	            "hook_event_name": "PostToolUse", "session_id": "s-exec",
    99	            "tool_name": tool, "tool_use_id": tool_id,
   100	            "tool_input": tool_input, "tool_response": response,
   101	        })
   102	
   103	    observations: dict[str, object] = {}
   104	    observations["explain_context"] = prompt("Why was Agent skipped? Explain only.", "s-explain")
   105	    observations["explain_blocks_bash"] = pre("Bash", {"command": "echo mutate"}, "s-explain")
   106	    observations["pause_context"] = prompt("stop the loop", "s-pause")
   107	    observations["pause_blocks_edit"] = pre("Edit", {"file_path": str(run / "frontier.md")}, "s-pause")
   108	    observations["execute_context"] = prompt("resume execution", "s-exec")
   109	    target = {"command": "python3 tools/probe.py GET https://example.test"}
   110	    contract_file = run / "state" / "turn_contract.json"
   111	    contract_file.write_text("{broken", encoding="utf-8")
   112	    observations["malformed_contract_blocks_target"] = pre("Bash", target)
   113	    prompt("resume execution", "s-exec")
   114	    contract_data = json.loads(contract_file.read_text(encoding="utf-8"))
   115	    contract_data["schema"] = "wrong"
   116	    contract_file.write_text(json.dumps(contract_data), encoding="utf-8")
   117	    observations["wrong_schema_blocks_target"] = pre("Bash", target)
   118	    prompt("resume execution", "s-exec")
   119	    original_frontier = (run / "frontier.md").read_text(encoding="utf-8")
   120	    (run / "frontier.md").write_text(
   121	        "# Frontier\n## Open Fronts\nmalformed\n", encoding="utf-8")
   122	    observations["malformed_frontier_blocks_target"] = pre("Bash", target)
   123	    (run / "frontier.md").write_text(original_frontier, encoding="utf-8")
   124	    observations["target_before_agents"] = pre("Bash", target)
   125	
   126	    assignments_path = run / "state" / "assignments.json"
   127	    original_assignments = assignments_path.read_text(encoding="utf-8")
   128	    assignments_path.write_text("{broken", encoding="utf-8")
   129	    observations["malformed_assignments_blocks_agent"] = pre(
   130	        "Agent", {"prompt": "XUNJI_ASSIGNMENT=A-web-001 XUNJI_FRONT=F-001"})
   131	    assignments_path.write_text(original_assignments, encoding="utf-8")
   132	
   133	    for tool_id, assignment, front in (
   134	        ("agent-tool-1", "A-web-001", "F-001"),
   135	        ("agent-tool-2", "A-auth-001", "F-002"),
   136	    ):
   137	        agent_input = {"prompt": f"XUNJI_ASSIGNMENT={assignment} XUNJI_FRONT={front}"}
   138	        observations[f"pre_{assignment}"] = pre("Agent", agent_input)
   139	        post("Agent", tool_id, agent_input, {"result": f"candidate for {front}"})
   140	
   141	    before_disposition = runtime_receipts.agent_disposition(run, session_id="s-exec")
   142	    for assignment, front in (("A-web-001", "F-001"), ("A-auth-001", "F-002")):
   143	        workers.update_agent_lifecycle(
   144	            run, assignment, status="merged",
   145	            note=f"Front: {front} candidate adjudicated", terminal=True,
   146	        )
   147	    after_disposition = runtime_receipts.agent_disposition(run, session_id="s-exec")
   148	    observations["agent_before_disposition"] = before_disposition
   149	    observations["agent_after_disposition"] = after_disposition
   150	    observations["target_after_agents"] = pre("Bash", target)
   151	
   152	    cron_input = {"prompt": f"/loop {run.name}"}
   153	    observations["cron_create_before_list"] = pre("CronCreate", cron_input)
   154	    post("CronList", "cron-list-1", {}, {"tasks": []})
   155	    observations["cron_create_after_empty_list"] = pre("CronCreate", cron_input)
   156	    post("CronCreate", "cron-create-1", cron_input, {"id": "job-observed-1"})
   157	    observations["duplicate_create_without_relist"] = pre("CronCreate", cron_input)
   158	    post("CronList", "cron-list-2", {}, {
   159	        "tasks": [{"id": "job-observed-1", "prompt": f"/loop {run.name}"}],
   160	    })
   161	    observations["delete_unrelated_job"] = pre("CronDelete", {"id": "job-unrelated"})
   162	    observations["delete_observed_job"] = pre("CronDelete", {"id": "job-observed-1"})
   163	    post("CronDelete", "cron-delete-1", {"id": "job-observed-1"}, {"deleted": True})
   164	    post("CronList", "cron-list-3", {}, {"tasks": []})
   165	    observations["cron_quiescent"] = runtime_receipts.cron_quiescent(run, session_id="s-exec")
   166	
   167	    generated_at = datetime.now(timezone.utc).isoformat()
   168	    review_input = {"command": f"python3 tools/peer_review.py {run} --into-run"}
   169	    post("Bash", "review-tool-1", review_input, {"stdout": "## Verdict: PASS"})
   170	    observations["foreground_review_observed"] = runtime_receipts.review_invocation_valid(
   171	        run, generated_at)
   172	
   173	    evidence_index = run / "evidence_index.json"
   174	    evidence_index.write_text('{"schema":"observation"}\n', encoding="utf-8")
   175	    evidence_hash = hashlib.sha1(evidence_index.read_bytes()).hexdigest()
   176	    weak_prompt = f"XUNJI_COMPLETION_REVIEW EVIDENCE_INDEX={evidence_hash}"
   177	    structured_prompt = (
   178	        f"{weak_prompt} "
   179	        "CHECKS=report_parity,severity_artifacts,reachable_frontier,review_ledger"
   180	    )
   181	    observations["completion_prompt_missing_checks"] = pre(
   182	        "Agent", {"prompt": weak_prompt})
   183	    observations["completion_prompt_structured"] = pre(
   184	        "Agent", {"prompt": structured_prompt})
   185	    post("Agent", "completion-weak-1", {"prompt": structured_prompt}, {"result": "PASS"})
   186	    observations["bare_pass_rejected"] = not runtime_receipts.completion_review_valid(
   187	        run, evidence_hash)
   188	    post("Agent", "completion-structured-1", {"prompt": structured_prompt}, {
   189	        "result": (
   190	            f"XUNJI_COMPLETION_VERDICT=PASS EVIDENCE_INDEX={evidence_hash} "
   191	            "CHECKS=report_parity,severity_artifacts,reachable_frontier,review_ledger"
   192	        ),
   193	    })
   194	    observations["structured_completion_accepted"] = runtime_receipts.completion_review_valid(
   195	        run, evidence_hash)
   196	    events, chain_errors = runtime_receipts.validate_chain(run)
   197	    observations["runtime_chain"] = {"events": len(events), "errors": chain_errors}
   198	    observations["run_status"] = json.loads((run / "state" / "run_status.json").read_text(encoding="utf-8"))
   199	
   200	    deny = lambda value: value.get("hookSpecificOutput", {}).get("permissionDecision") == "deny"
   201	    checks = {
   202	        "sessionstart_contract_selftest_wired": any(
   203	            "tools/turn_contract.py" in command and "--selftest" in command
   204	            for command in session_start_commands),
   205	        "explain_denies_bash": deny(observations["explain_blocks_bash"]),
   206	        "pause_denies_edit": deny(observations["pause_blocks_edit"]),
   207	        "pre_agent_target_denied": deny(observations["target_before_agents"]),
   208	        "malformed_contract_denied": deny(observations["malformed_contract_blocks_target"]),
   209	        "wrong_contract_schema_denied": deny(observations["wrong_schema_blocks_target"]),
   210	        "malformed_frontier_denied": deny(observations["malformed_frontier_blocks_target"]),
   211	        "malformed_assignments_denied": deny(observations["malformed_assignments_blocks_agent"]),
   212	        "unmerged_agents_detected": bool(before_disposition.get("pending")),
   213	        "merged_agents_accepted": after_disposition.get("disposition_satisfied") is True,
   214	        "post_agent_target_allowed": observations["target_after_agents"] == {},
   215	        "cron_requires_list": deny(observations["cron_create_before_list"]),
   216	        "cron_allows_after_empty_list": observations["cron_create_after_empty_list"] == {},
   217	        "duplicate_cron_denied": deny(observations["duplicate_create_without_relist"]),
   218	        "unrelated_delete_denied": deny(observations["delete_unrelated_job"]),
   219	        "listed_delete_allowed": observations["delete_observed_job"] == {},
   220	        "cron_final_quiescent": observations["cron_quiescent"][0] is True,
   221	        "review_invocation_time_bound": observations["foreground_review_observed"] is True,
   222	        "completion_prompt_requires_checks": deny(observations["completion_prompt_missing_checks"]),
   223	        "completion_prompt_with_checks_allowed": observations["completion_prompt_structured"] == {},
   224	        "bare_completion_pass_rejected": observations["bare_pass_rejected"] is True,
   225	        "structured_completion_accepted": observations["structured_completion_accepted"] is True,
   226	        "receipt_chain_valid": not chain_errors and len(events) == 21,
   227	        "execute_status_active": observations["run_status"].get("status") == "active",
   228	    }
   229	    print(json.dumps({
   230	        "schema": "xunji.runtime_observation.v1",
   231	        "checks": checks,
   232	        "observations": observations,
   233	        "runtime_events": events,
   234	    }, ensure_ascii=False, indent=2, sort_keys=True))
   235	    return 0 if all(checks.values()) else 1
   236	
   237	
   238	if __name__ == "__main__":
   239	    raise SystemExit(main())
